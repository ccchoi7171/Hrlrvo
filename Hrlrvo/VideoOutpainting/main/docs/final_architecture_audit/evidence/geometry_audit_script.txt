@lru_cache(None)
def visible_bounds(r: Rect):
    # The same official latent-bound -> pixel-bound transform used by LTX positional geometry.
    coords=torch.tensor([[[[0,1]],[[r.y0,r.y1]],[[r.x0,r.x1]]]],dtype=torch.int64)
    pixel=get_pixel_coords(coords,sf,causal_fix=True)[0,:,0,:]
    return (int(pixel[2,0])-boundary.left,int(pixel[1,0])-boundary.top,
            int(pixel[2,1])-boundary.left,int(pixel[1,1])-boundary.top)

def xyxy(r): return list(asdict(r).values()) if r else None

def intersects(a,b):
    y0,x0,y1,x1=max(a.y0,b.y0),max(a.x0,b.x0),min(a.y1,b.y1),min(a.x1,b.x1)
    return Rect(y0,x0,y1,x1) if y0<y1 and x0<x1 else None

def rb(r): return f'[{r.y0}:{r.y1}, {r.x0}:{r.x1})' if r else 'N/A'

def area(r): return r.height*r.width if r else 0

def pointbox(points):
    return Rect(min(y for y,x in points),min(x for y,x in points),max(y for y,x in points)+1,max(x for y,x in points)+1) if points else None

def direction(box,support,axis):
    if not support:return 'N/A'
    if axis=='x': lo,hi=box.x0-support.x0,support.x1-box.x1; names=('left','right')
    else: lo,hi=box.y0-support.y0,support.y1-box.y1; names=('up','down')
    return '+'.join(f'{n}:{v}' for n,v in zip(names,(lo,hi)) if v) or 'none'

def draw_rect(draw,r,fill=None,outline=None,width=1):
    a,b,c,d=visible_bounds(r)
    # Image display crop only; canonical geometry is never changed.
    a,b,c,d=max(a,0),max(b,0),min(c,VW),min(d,VH)
    if a<c and b<d: draw.rectangle((a,b,c-1,d-1),fill=fill,outline=outline,width=width)

def dashed(draw,r,color,width=5):
    a,b,c,d=visible_bounds(r); a,b,c,d=max(a,0),max(b,0),min(c-1,VW-1),min(d-1,VH-1)
    for x in range(a,c,28):
        draw.line((x,b,min(x+15,c),b),fill=color,width=width); draw.line((x,d,min(x+15,c),d),fill=color,width=width)
    for y in range(b,d,28):
        draw.line((a,y,a,min(y+15,d)),fill=color,width=width); draw.line((c,y,c,min(y+15,d)),fill=color,width=width)
