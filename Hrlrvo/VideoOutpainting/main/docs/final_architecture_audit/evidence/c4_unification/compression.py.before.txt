"""UltraGen spatial depthwise C/R, extended to exact, unpadded footprints.

Upstream 4f9ec33 has *no* custom C/R initializer. Hrlrvo explicitly initializes
C as an average and R's refinement as identity; their operators stay unchanged.
"""

from collections import defaultdict

import torch
from torch import nn
from torch.nn import functional as F


class SpatialCompressor(nn.Module):
    def __init__(self, dim: int, scale: int):
        super().__init__()
        self.scale = scale
        self.conv = nn.Conv2d(dim, dim, kernel_size=scale, stride=scale, padding=0, groups=dim)
        nn.init.constant_(self.conv.weight, 1.0 / (scale * scale))
        nn.init.zeros_(self.conv.bias)

    def forward(self, grid: torch.Tensor, cells: tuple) -> torch.Tensor:
        """[B,T,H,W,D] -> [B,T,N,D] over explicit canonical cell footprints.

        Equal support shapes are batched. Full cells execute the actual stride-s
        depthwise Conv2d. Adaptive supplies only full C2/C4 cells (which may
        contain multiple pixel roles). A Global canvas-edge cell uses valid taps
        and normalizes by its represented area, exactly matching average pooling
        at initialization. No zero-filled spatial cells are manufactured.
        """
        b, t, _, _, d = grid.shape
        if not cells:
            return grid.new_empty(b, t, 0, d)
        buckets = defaultdict(list)
        for i, cell in enumerate(cells):
            if cell.scale != self.scale:
                raise ValueError("A compressor may only consume cells at its own density")
            buckets[cell.offsets].append((i, cell.indices))
        flat = grid.flatten(2, 3)
        result = grid.new_empty(b, t, len(cells), d)
        for offsets, items in buckets.items():
            order = torch.tensor([i for i, _ in items], device=grid.device)
            indices = torch.tensor([indices for _, indices in items], device=grid.device)
            values = flat[:, :, indices, :]
            if len(offsets) == self.scale**2:
                spatial = values.permute(0, 1, 2, 4, 3).reshape(-1, d, self.scale, self.scale)
                out = self.conv(spatial).reshape(b, t, len(items), d)
            else:
                taps = torch.tensor(offsets, device=grid.device)
                weight = self.conv.weight.flatten(1).index_select(1, taps).to(values.dtype)
                out = torch.einsum("btnkc,ck->btnc", values, weight)
                out = out * (self.scale**2 / len(offsets)) + self.conv.bias.to(out.dtype)
            result = result.index_copy(2, order, out.to(result.dtype))
        return result


class SpatialDecompressor(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        # UltraGen: bilinear torchvision Resize then depthwise 3D refinement.
        # Hrlrvo identity initialization preserves the upsampled feature at start;
        # all temporal/spatial taps remain trainable, including off-center taps.
        self.refine = nn.Conv3d(dim, dim, kernel_size=3, stride=1, padding=1, groups=dim)
        nn.init.zeros_(self.refine.weight)
        nn.init.ones_(self.refine.weight[:, :, 1, 1, 1])
        nn.init.zeros_(self.refine.bias)

    def forward(self, tokens, compressed_shape, target_shape):
        b, t, n, d = tokens.shape
        ch, cw = compressed_shape
        h, w = target_shape
        if n != ch * cw:
            raise ValueError("Decompression requires the exact rectangular target-token map")
        spatial = tokens.reshape(b * t, ch, cw, d).permute(0, 3, 1, 2)
        # torchvision Resize's tensor default: bilinear, align_corners=False,
        # antialias=True. Float32 also supports bf16 inputs on CPU backends.
        spatial = F.interpolate(spatial.float(), size=(h, w), mode="bilinear",
                                align_corners=False, antialias=True).to(tokens.dtype)
        volume = spatial.reshape(b, t, d, h, w).permute(0, 2, 1, 3, 4)
        return self.refine(volume).permute(0, 2, 3, 4, 1)
